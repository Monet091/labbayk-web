    <script>
        // 1. SIMULASI DATA DATABASE (Nanti ini diganti dengan URL API asli Anda)
        // Contoh: const API_URL = "http://localhost/api/get_absensi.php";
        const API_URL = "simulasi-database"; 

        // Data tiruan yang mensimulasikan isi database Anda
        const dummyDatabase = [
            { id: 101, nama: "Claudia Zera Hehamony", divisi: "Marketing", status: "Disetujui" },
            { id: 102, nama: "Kim Jaeyun", divisi: "Marketing", status: "Tidak Disetujui" },
            { id: 103, nama: "Kim Mingyu", divisi: "IT", status: "Disetujui" },
            { id: 104, nama: "Alliya Nasywa Aulia Ruswan", divisi: "Content Creator", status: "Disetujui" }
        ];

        // 2. FUNGSI UNTUK MENGAMBIL DATA (FETCH)
        async function ambilDataAbsen() {
            try {
                let data;

                if (API_URL === "simulasi-database") {
                    // Mensimulasikan jeda jaringan selama 1 detik agar terasa seperti fetch asli
                    data = await new Promise(resolve => setTimeout(() => resolve(dummyDatabase), 1000));
                } else {
                    // KODE ASLI: Aktifkan ini jika sudah ada url backend asli
                    const response = await fetch(API_URL);
                    if (!response.ok) throw new Error("Gagal mengambil data dari server");
                    data = await response.json();
                }

                // Jalankan fungsi untuk merender data ke HTML
                renderTabel(data);

            } catch (error) {
                console.error("Error:", error);
                document.getElementById('tabel-absen').innerHTML = `
                    <tr>
                        <td colspan="4" class="text-center p-4 text-red-500 font-semibold">
                            Gagal memuat data. Silakan coba lagi.
                        </td>
                    </tr>
                `;
            }
        }

        // 3. FUNGSI UNTUK MERENDER DATA KEDALAM STRUKTUR TABEL HTML
        function renderTabel(kumpulanData) {
            const tbody = document.getElementById('tabel-absen');
            
            // Bersihkan isi tabel (menghapus tulisan "Memuat data...")
            tbody.innerHTML = ""; 

            // Jika data kosong
            if (kumpulanData.length === 0) {
                tbody.innerHTML = `<tr><td colspan="4" class="text-center p-4">Tidak ada pengajuan izin.</td></tr>`;
                return;
            }

            // Lakukan perulangan (looping) untuk setiap data di database
            kumpulanData.forEach(item => {
                // Membuat elemen baris (tr) baru
                const baris = document.createElement('tr');
                baris.className = "border border-[#7c7c7c] hover:bg-gray-50/50 transition-colors";

                // Mengisi struktur td di dalam baris tersebut dengan padding dinamis (p-2 di HP, md:p-3 di Laptop)
                baris.innerHTML = `
                    <td class="border border-[#7c7c7c] p-2 md:p-3 whitespace-nowrap">${item.nama}</td>
                    <td class="border border-[#7c7c7c] p-2 md:p-3">${item.divisi}</td>
                    <td class="border border-[#7c7c7c] p-2 md:p-3 text-[#4CAF50] font-semibold whitespace-nowrap">
                        <span class="inline-flex items-center">
                            ${item.status} 
                            <span class="inline-block w-2 h-2 bg-[#b7b7b7] rounded-full ml-1.5"></span>
                        </span>
                    </td>
                    <td class="border border-[#7c7c7c] p-2 md:p-3 text-center">
                        <a href="detail_form.html?id=${item.id}" class="text-[#0000ee] underline font-medium hover:text-blue-800">FORM</a>
                    </td>
                `;

                // Masukkan baris baru tersebut ke dalam tbody HTML
                tbody.appendChild(baris);
            });
        }

        // 4. JALANKAN FUNGSI KETIKA HALAMAN SELESAI DIMUAT
        document.addEventListener("DOMContentLoaded", ambilDataAbsen);
    </script>
